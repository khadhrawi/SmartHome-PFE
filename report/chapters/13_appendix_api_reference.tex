\chapter{Appendix A: Complete API Reference}

\newcommand{\route}[1]{\path{#1}}

\section{Usage Notes}
This appendix serves as a practical API handbook for developers, testers, and maintainers. Routes are grouped by functional domain. All protected routes require the Authorization header:
\begin{lstlisting}[style=codeblock]
Authorization: Bearer <JWT_TOKEN>
\end{lstlisting}

\section{Authentication API}

\subsection{Resident Registration}
\textbf{Endpoint}: POST /api/auth/resident/register

\textbf{Request payload}
\begin{lstlisting}[style=codeblock,language=JavaScript]
{
  "name": "Resident Name",
  "email": "resident@example.com",
  "password": "StrongPass#123",
  "houseCode": "H1234",
  "assignedRoom": "bedroom",
  "roomRequest": "",
  "inviteCode": "OPTIONAL"
}
\end{lstlisting}

\textbf{Success response}
\begin{lstlisting}[style=codeblock,language=JavaScript]
{
  "_id": "...",
  "name": "Resident Name",
  "email": "resident@example.com",
  "role": "resident",
  "houseCode": "H1234",
  "permissions": [],
  "token": "..."
}
\end{lstlisting}

\section{Representative Error Catalog}
\begin{longtable}{p{4cm}p{2cm}p{8cm}}
\toprule
Error Condition & HTTP & Message Pattern \\
\midrule
Missing or invalid token & 401 & Not authorized, token failed/no token \\
Role denied & 403 & Not authorized as an admin/agency account \\
Resource not found & 404 & Device/Unit/Request not found \\
Validation failure & 400/422 & Invalid field, weak password, invalid code \\
Conflict & 409 & Similar pending request already exists \\
\bottomrule
\end{longtable}

\midrule
