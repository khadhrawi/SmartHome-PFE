\section{Routage de l'Application}

L'application React utilise \textbf{React Router v7} avec des gardes de routes basées sur
les rôles. Les routes sont protégées par l'\texttt{AuthContext} qui lit le JWT depuis
localStorage et expose l'objet utilisateur courant à tous les composants.

\begin{lstlisting}[style=codeblock,language=JavaScript,caption={Configuration des Routes (frontend/src/App.jsx)}]
<Routes>
  {/* Routes publiques */}
  <Route path="/"              element={<Home />} />
  <Route path="/auth"          element={<AuthEntry />} />
  <Route path="/admin/login"   element={<AdminLogin />} />
  <Route path="/admin/register"  element={<AdminRegister />} />
  <Route path="/resident/login"  element={<ResidentLogin />} />
  <Route path="/oauth/callback"  element={<OAuthCallback />} />
  <Route path="/verify-email"    element={<VerifyEmail />} />
  <Route path="/reset-password"  element={<ResetPassword />} />

  {/* Protegees : routes Admin */}
  <Route element={<AdminGuard />}>
    <Route path="/dashboard"       element={<Dashboard />} />
    <Route path="/devices"         element={<Devices />} />
    <Route path="/rooms"           element={<Rooms />} />
    <Route path="/scenarios"       element={<Scenarios />} />
    <Route path="/security"        element={<Security />} />
    <Route path="/energy"          element={<Energy />} />
    <Route path="/users"           element={<UserManagement />} />

  {/* Protegees : routes Resident */}

  <Route element={<AgencyGuard />}>
    <Route path="/agency/dashboard" element={<AgencyDashboard />} />

  <Route path="/concierge-hub" element={<ConciergeHub />} />

\section{Implémentation AuthContext}

\begin{lstlisting}[style=codeblock,language=JavaScript,caption={AuthContext (frontend/src/context/AuthContext.jsx)}]
const AuthContext = createContext(null);

export const AuthProvider = ({ children }) => {
  const [user,  setUser]  = useState(null);
  const [token, setToken] = useState(
    () => localStorage.getItem('token')
  );

  // Charger l'utilisateur si token present
  useEffect(() => {
    if (!token) return;
    axios.get('/api/auth/me', {
      headers: { Authorization: `Bearer ${token}` }
    })
      .then(({ data }) => setUser(data.user))
      .catch(() => {
        setToken(null);
        localStorage.removeItem('token');
      });
  }, [token]);

  const login = (newToken, newUser) => {
    localStorage.setItem('token', newToken);
    setToken(newToken);
    setUser(newUser);
  };

  const logout = () => {
    localStorage.removeItem('token');
    setToken(null);
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, token, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => useContext(AuthContext);
\end{lstlisting}

\section{ThemeContext et Mode Sombre}

L'application supporte le basculement thème sombre/clair via la stratégie de classe
\texttt{dark} de \textbf{Tailwind CSS}. Le thème sélectionné est stocké dans
\texttt{localStorage} et appliqué en basculant \texttt{document.documentElement.classList}.

\begin{lstlisting}[style=codeblock,language=JavaScript,caption={Impl\'ementation ThemeContext (frontend/src/context/ThemeContext.jsx)}]
const ThemeContext = createContext(null);

export const ThemeProvider = ({ children }) => {
  const [dark, setDark] = useState(
    () => localStorage.getItem('theme') === 'dark'
  );

  useEffect(() => {
    document.documentElement.classList.toggle('dark', dark);
    localStorage.setItem('theme', dark ? 'dark' : 'light');
  }, [dark]);

  return (
    <ThemeContext.Provider value={{ dark, toggle: () => setDark(d => !d) }}>
      {children}
    </ThemeContext.Provider>
  );
};
\end{lstlisting}
